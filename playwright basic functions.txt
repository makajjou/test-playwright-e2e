LoginPage.ts
import { type Locator, type Page } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly usernameInput: Locator;
  readonly passwordInput: Locator;
  readonly loginButton: Locator;
  readonly errorMessage: Locator;

  constructor(page: Page) {
    this.page = page;

    this.usernameInput = page.getByLabel('Utilisateur');
    this.passwordInput = page.getByLabel('Mot de passe');

    this.loginButton = page.getByRole('button', {
      name: 'Se connecter',
    });

    this.errorMessage = page.getByRole('alert');
  }

  async goto(): Promise<void> {
    await this.page.goto('https://monsite.com/login');
  }

  async login(username: string, password: string): Promise<void> {
    await this.usernameInput.fill(username);
    await this.passwordInput.fill(password);
    await this.loginButton.click();
  }
}

import { type Locator, type Page } from '@playwright/test';

export class HeaderComponent {
  readonly page: Page;
  readonly accountButton: Locator;
  readonly logoutButton: Locator;
  readonly homeLink: Locator;
  readonly logo: Locator;

  constructor(page: Page) { // équivalent à init en python, pour initier les attributs
    this.page = page;

    this.accountButton = page.getByRole('button', {
      name: 'Mon compte',
    });

    this.logoutButton = page.getByRole('button', {
      name: 'Se déconnecter',
    });

    this.homeLink = page.getByRole('link', {
      name: 'Accueil',
    });

    this.logo = page.getByRole('img', {
      name: 'Logo',
    });
  }

  async openAccountMenu(): Promise<void> { //On définit une méthode qui a comme sortie le vide (elle fait juste une promesse qui retourne du vide)
    await this.accountButton.click();
  }

  async logout(): Promise<void> {
    await this.accountButton.click();
    await this.logoutButton.click();
  }

  async goToHome(): Promise<void> {
    await this.homeLink.click();
  }
}
